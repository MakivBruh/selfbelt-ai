import { useState } from 'react'
import { Section } from './Layout'
import { useCopy, useSimulation } from '../simulation/context'

export function Hero() {
  const t = useCopy()
  return <section id="hero" aria-labelledby="hero-heading" className="py-8 md:py-12">
    <p className="text-xs font-semibold tracking-widest text-accent">{t.prototype}</p>
    <h1 id="hero-heading" className="mt-3 text-3xl font-semibold md:text-4xl">{t.title}</h1>
    <p className="mt-3 max-w-2xl text-muted">{t.intro}</p>
    <p className="mt-3 text-sm text-muted">{t.disclaimer}</p>
  </section>
}
export function Dashboard() {
  const t = useCopy()
  const { state } = useSimulation()
  const stats = [[t.speed, `${state.speedKmh} km/h`], [t.occupants, String(Object.values(state.seats).filter(s => s.occupied).length)], [t.source, t.manual]]
  return <Section id="dashboard" title={t.dashboard}><dl className="grid gap-5 sm:grid-cols-3">
    {stats.map(([label, value]) => <div key={label}><dt className="text-sm text-muted">{label}</dt><dd className="mt-2 text-xl font-semibold tabular-nums">{value}</dd></div>)}
  </dl></Section>
}
function Placeholder({ note }: { note: string }) {
  const t = useCopy()
  return <div className="flex min-h-40 flex-col justify-center gap-3 rounded border border-dashed border-border p-5">
    <p className="text-xs font-medium uppercase tracking-wide text-accent">{t.placeholder}</p><p className="max-w-xl text-sm leading-relaxed text-muted">{note}</p>
  </div>
}
export function TopView() { const t = useCopy(); return <Section id="top-view" title={t.top}><Placeholder note={t.topNote} /></Section> }
export function SideView() { const t = useCopy(); return <Section id="side-view" title={t.side}><Placeholder note={t.sideNote} /></Section> }
export function DetectionPanel() { const t = useCopy(); return <Section id="detection" title={t.detection}><Placeholder note={t.detectionNote} /></Section> }
export function ScenarioPlayground() {
  const t = useCopy()
  const { state, dispatch } = useSimulation()
  const [didReset, setDidReset] = useState(false)
  return <Section id="playground" title={t.playground}>
    <label htmlFor="speed" className="flex justify-between gap-4 text-sm">{t.speed}<output htmlFor="speed" className="font-mono text-accent">{state.speedKmh} km/h</output></label>
    <input id="speed" type="range" min="0" max="120" step="5" value={state.speedKmh} onChange={e => { setDidReset(false); dispatch({ type: 'setSpeed', value: Number(e.target.value) }) }} className="my-4 min-h-11 w-full accent-accent" />
    <p className="text-sm text-muted">{t.playgroundNote}</p>
    <button type="button" onClick={() => { dispatch({ type: 'reset' }); setDidReset(true) }} className="mt-5 min-h-11 rounded border border-border px-4 text-sm hover:bg-slate-800">{t.reset}</button>
    <p role="status" className="mt-2 min-h-5 text-sm text-muted">{didReset ? t.resetDone : ''}</p>
  </Section>
}
export function SystemFlow() {
  const t = useCopy()
  return <Section id="system-flow" title={t.flow}><ol className="grid gap-4 sm:grid-cols-2 lg:grid-cols-5">
    {t.flowSteps.map((step, i) => <li key={i} className="border-t border-border pt-3 text-sm"><span className="mb-2 block font-mono text-accent">0{i + 1}</span>{step}</li>)}
  </ol></Section>
}
export function Footer() { const t = useCopy(); return <footer className="mt-8 border-t border-border py-6 text-sm text-muted">{t.footer}</footer> }

src/simulation/model.ts
=======================
export type Language = 'id' | 'en'
export type SeatId = 'driver' | 'rearLeft' | 'rearRight'
export type SeatState =
  | { occupied: false; belted: false }
  | { occupied: true; belted: boolean }
export type SimulationState = { speedKmh: number; seats: Record<SeatId, SeatState> }
export type SimulationAction =
  | { type: 'setSpeed'; value: number }
  | { type: 'setSeat'; seat: SeatId; value: SeatState }
  | { type: 'reset' }
export function createInitialState(): SimulationState {
  return { speedKmh: 0, seats: {
    driver: { occupied: true, belted: true },
    rearLeft: { occupied: false, belted: false },
    rearRight: { occupied: false, belted: false },
  } }
}
export function simulationReducer(state: SimulationState, action: SimulationAction): SimulationState {
  switch (action.type) {
    case 'reset': return createInitialState()
    case 'setSpeed':
      if (!Number.isFinite(action.value) || action.value < 0 || action.value > 120) return state
      if (action.value > 0 && !state.seats.driver.occupied) return state
      return { ...state, speedKmh: action.value }
    case 'setSeat':
      if (!Object.hasOwn(state.seats, action.seat)) return state
      if (action.seat === 'driver' && !action.value.occupied && state.speedKmh > 0) return state
      return { ...state, seats: { ...state.seats, [action.seat]: action.value.occupied
        ? { occupied: true, belted: action.value.belted }
        : { occupied: false, belted: false } } }
  }
}

src/simulation/context.ts

import { createContext, useContext } from 'react'
import type { Dispatch } from 'react'
import type { Language, SimulationAction, SimulationState } from './model'
import { translations } from '../i18n/translations'
type SimulationContextValue = {
  state: SimulationState
  dispatch: Dispatch<SimulationAction>
  language: Language
  setLanguage: (language: Language) => void
}
export const SimulationContext = createContext<SimulationContextValue | null>(null)
export function useSimulation() {
  const context = useContext(SimulationContext)
  if (!context) throw new Error('useSimulation requires SimulationProvider')
  return context
}
export function useCopy() { return translations[useSimulation().language] }

src/simulation/SimulationProvider.tsx
=====================================
import { useEffect, useReducer, useState } from 'react'
import type { PropsWithChildren } from 'react'
import { createInitialState, simulationReducer } from './model'
import type { Language } from './model'
import { SimulationContext } from './context'
export function SimulationProvider({ children }: PropsWithChildren) {
  const [state, dispatch] = useReducer(simulationReducer, undefined, createInitialState)
  const [language, setLanguage] = useState<Language>('id')
  useEffect(() => { document.documentElement.lang = language }, [language])
  return <SimulationContext.Provider value={{ state, dispatch, language, setLanguage }}>{children}</SimulationContext.Provider>
}

import assert from 'node:assert/strict'
import { createInitialState, simulationReducer } from '../src/simulation/model.ts'
const initial = createInitialState()
const moving = simulationReducer(initial, { type: 'setSpeed', value: 50 })
assert.equal(moving.speedKmh, 50)
assert.equal(initial.speedKmh, 0)
for (const value of [NaN, Infinity, -1, 121]) assert.equal(simulationReducer(initial, { type: 'setSpeed', value }), initial)
assert.equal(simulationReducer(moving, { type: 'setSeat', seat: 'driver', value: { occupied: false, belted: false } }), moving)
const empty = simulationReducer(initial, { type: 'setSeat', seat: 'driver', value: { occupied: false, belted: false } })
assert.equal(simulationReducer(empty, { type: 'setSpeed', value: 5 }), empty)
assert.deepEqual(simulationReducer(moving, { type: 'reset' }), initial)
console.log('PASS: immutable updates, speed bounds, driver constraints, reset')
